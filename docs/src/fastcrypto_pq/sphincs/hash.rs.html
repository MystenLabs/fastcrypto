4/// FIPS 205 Section 11.2.1 — SHA2-based tweakable hash functions for security category 1 (n=16).
5///
6/// F, H, T_l, and PRF all share the same core:
7///   Trunc_n(SHA-256(PK.seed ‖ toByte(0, 64-n) ‖ ADRSc ‖ M))
8///
9/// TODO: this only covers the n=16 SHA2 instantiation. Extend to cover:
10///   - n ≥ 24 (SHA2-192/256): H and T_l switch to SHA-512 with 128-byte block
11///     pad (toByte(0, 128-n)); F and PRF stay on SHA-256. See FIPS 205 §11.2.1.
12///   - SLH-DSA-SHAKE-*: the SHAKE256-based construction from §11.2.2, which
13///     replaces the whole family with a single XOF — no SHA-256/SHA-512 split,
14///     no 64-n padding.
15use digest::Digest;
16use sha2::Sha256;
17
18use super::Adrs;
19
20const SHA256_BLOCK: usize = 64;
21const ZERO_PAD: [u8; SHA256_BLOCK] = [0u8; SHA256_BLOCK];
22
23/// Core shared by F, H, T_l, and PRF:
24///   Trunc_n(SHA-256(PK.seed ‖ toByte(0, 64-n) ‖ ADRSc ‖ M))
25pub fn tweakable_hash(pk_seed: &[u8], adrs: Adrs, m: &[u8]) -> Vec<u8> {
26    let n = pk_seed.len();
27    let adrs_c = adrs.compress();
28    let mut hasher = Sha256::new();
29    // TODO: the first block is fixed - so we can share its output across instantiations to optimize hash costs
30    hasher.update(pk_seed);
31    hasher.update(&ZERO_PAD[..SHA256_BLOCK - n]);
32    hasher.update(adrs_c);
33    hasher.update(m);
34    hasher.finalize()[..n].to_vec()
35}
