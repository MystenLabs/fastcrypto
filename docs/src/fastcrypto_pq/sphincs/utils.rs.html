4/// Unpack each byte of `bytes` into individual bits (MSB first), each stored as 0 or 1.
5pub(super) fn bytes_to_bits(bytes: &[u8]) -> Vec<u8> {
6    let mut bits = Vec::with_capacity(bytes.len() * 8);
7    for &b in bytes {
8        for shift in (0..8).rev() {
9            bits.push((b >> shift) & 1);
10        }
11    }
12    bits
13}
14
15/// Pack `bits` into `chunk_size`-bit values (MSB first within each chunk).
16pub(super) fn bits_to_base(bits: &[u8], chunk_size: u16) -> Vec<u32> {
17    assert!(chunk_size <= 32, "chunk_size must fit in u32");
18    let n = chunk_size as usize;
19    assert_eq!(
20        bits.len() % n,
21        0,
22        "bits.len() must be a multiple of chunk_size"
23    );
24    let mut result = Vec::with_capacity(bits.len() / n);
25    for chunk in bits.chunks(n) {
26        let mut acc: u32 = 0;
27        for &b in chunk {
28            acc = (acc << 1) | u32::from(b);
29        }
30        result.push(acc);
31    }
32    result
33}

35#[cfg(test)]
36mod tests {
37    use super::*;
38
39    #[test]
40    fn test_bits_to_base_nibbles() {
41        assert_eq!(bits_to_base(&bytes_to_bits(&[0xA3]), 4), vec![0xA, 0x3]);
42        assert_eq!(
43            bits_to_base(&bytes_to_bits(&[0xA3, 0xF0]), 4),
44            vec![0xA, 0x3, 0xF, 0x0]
45        );
46    }
47
48    #[test]
49    fn test_bits_to_base_pairs() {
50        // chunk_size = 2 → 0xA3 = 10 10 00 11 → [2, 2, 0, 3]
51        assert_eq!(bits_to_base(&bytes_to_bits(&[0xA3]), 2), vec![2, 2, 0, 3]);
52    }
53
54    #[test]
55    fn test_bits_to_base_single_bits() {
56        assert_eq!(
57            bits_to_base(&bytes_to_bits(&[0xA3]), 1),
58            vec![1, 0, 1, 0, 0, 0, 1, 1]
59        );
60    }
