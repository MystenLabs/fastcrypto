4//! Shared Merkle-tree helpers used by both XMSS (Tree ADRS) and FORS (FORS_TREE ADRS).
5//!
6//! Three operations show up identically in both constructions:
7//!   - Compute a node at `(height z, index i)` by recursing to leaves.
8//!   - Build the authentication path from a leaf up to the root.
9//!   - Fold a leaf up to a root along an authentication path.
10//!
11//! Each caller supplies the leaf producer and an ADRS builder for interior nodes;
12//! everything else — parity-based sibling order, index shifting, hashing — is shared.
13
14use crate::sphincs::hash::tweakable_hash;
15use crate::sphincs::Adrs;
16
17/// Recursively compute the node at height `z`, index `i`.
18///   - `leaf(i)` produces the leaf value at global index `i` (called only at `z == 0`).
19///   - `mk_interior_adrs(height, index)` builds the ADRS used to hash an interior
20///     node at the given (parent) height and index.
21pub fn merkle_node(
22    leaf: &impl Fn(u32) -> Vec<u8>,
23    mk_interior_adrs: &impl Fn(u16, u32) -> Adrs,
24    i: u32,
25    z: u16,
26    pk_seed: &[u8],
27) -> Vec<u8> {
28    if z == 0 {
29        leaf(i)
30    } else {
31        let lnode = merkle_node(leaf, mk_interior_adrs, 2 * i, z - 1, pk_seed);
32        let rnode = merkle_node(leaf, mk_interior_adrs, 2 * i + 1, z - 1, pk_seed);
33        tweakable_hash(pk_seed, mk_interior_adrs(z, i), &[lnode, rnode].concat())
34    }
35}
36
37/// Build a length-`h` authentication path for leaf `leaf_idx` in a Merkle tree.
38/// `node_at(height, local_sibling_idx)` supplies the sibling node at each step.
39pub fn build_auth_path<F>(h: u16, leaf_idx: u32, node_at: F) -> Vec<Vec<u8>>
40where
41    F: Fn(u16, u32) -> Vec<u8>,
42{
43    let mut auth = Vec::with_capacity(h as usize);
44    let mut idx = leaf_idx;
45    for height in 0..h {
46        let sibling = idx ^ 1;
47        auth.push(node_at(height, sibling));
48        idx /= 2;
49    }
50    auth
51}
52
53/// Fold `leaf` up to the root by consuming each sibling in `auth`. At each step,
54/// the parity of `idx` decides sibling order; `mk_interior_adrs` builds the ADRS
55/// for the parent node at the new height/index.
56pub fn compute_root_from_path(
57    leaf: Vec<u8>,
58    mut idx: u32,
59    auth: &[Vec<u8>],
60    pk_seed: &[u8],
61    mk_interior_adrs: impl Fn(u16, u32) -> Adrs,
62) -> Vec<u8> {
63    let mut cur = leaf;
64    for (h, sibling) in auth.iter().enumerate() {
65        let combined = if idx.is_multiple_of(2) {
66            [cur.as_slice(), sibling.as_slice()].concat()
67        } else {
68            [sibling.as_slice(), cur.as_slice()].concat()
69        };
70        idx /= 2;
71        cur = tweakable_hash(pk_seed, mk_interior_adrs((h + 1) as u16, idx), &combined);
72    }
73    cur
74}
