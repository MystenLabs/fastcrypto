4//! SLH-DSA Hypertree implementation from FIPS-205
5//!
6//! A hypertree is `d` stacked XMSS trees. Layer 0's WOTS+ leaves sign the
7//! message; each layer above signs the root of the layer below; the top
8//! layer's root is the SLH-DSA public key root.
9
10use crate::sphincs::xmss::{xmss_node, xmss_pk_from_sig, xmss_sign, XmssParams, XmssSignature};
11use crate::sphincs::Adrs;
12
13// ------------------------------------------------------------------
14//                            Parameters
15// ------------------------------------------------------------------
16
17pub struct HypertreeParams {
18    /// Number of layers of XMSS trees.
19    pub d: u8,
20    pub xmss: XmssParams,
21    /// Total height of the hypertree (`h = d * h'`).
22    pub h: u8,
23}
24
25impl HypertreeParams {
26    pub fn new(d: u8, xmss: XmssParams) -> Self {
27        // h' < 32 so that a leaf index fits in u32 (the type of idx_leaf in
28        // the sign/verify API). FIPS 205 tops out at h' = 12 and SP 800-230
29        // draft at h' = 22, so this is purely a sanity check.
30        debug_assert!(xmss.h_prime < 32, "h_prime must be < 32");
31        // h = d * h' ≤ 68 for every FIPS 205 / SP 800-230 param set, so fits in u8.
32        let h = d
33            .checked_mul(xmss.h_prime)
34            .expect("h = d * h_prime overflowed u8");
35        HypertreeParams { d, xmss, h }
36    }
37}
38
39pub struct HypertreeSignature {
40    /// One XMSS signature per layer; `xmss[0]` is the bottom layer.
41    pub xmss: Vec<XmssSignature>,
42}
43
44impl HypertreeSignature {
45    pub fn size_in_bytes(&self) -> usize {
46        self.xmss.iter().map(|s| s.size_in_bytes()).sum()
47    }
48}
49
50// ------------------------------------------------------------------
51//                             Helpers
52// ------------------------------------------------------------------
53
54/// Consume the low `h_prime` bits of `idx_tree` as the next layer's `idx_leaf`,
55/// and shift `idx_tree` down by `h_prime`.
56fn advance_indices(idx_tree: &mut u64, idx_leaf: &mut u32, h_prime: u8) {
57    *idx_leaf = (*idx_tree & ((1u64 << h_prime) - 1)) as u32;
58    *idx_tree >>= h_prime;
59}
60
61// ------------------------------------------------------------------
62//                          Public API
63// ------------------------------------------------------------------
64
65/// Top-tree root (layer d-1, tree 0). This is the SLH-DSA public key root.
66pub fn ht_pk_root(params: &HypertreeParams, sk_seed: &[u8], pk_seed: &[u8]) -> Vec<u8> {
67    let adrs = Adrs::new().with_xmss_height(params.d - 1);
68    xmss_node(&params.xmss, sk_seed, 0, params.xmss.h_prime, pk_seed, adrs)
69}
70
71/// Sign `msg` at the layer-0 leaf identified by `(idx_tree, idx_leaf)`:
72/// - `idx_tree`: which XMSS tree at the current layer (upper `h - h'` bits of the global leaf index on entry to layer 0).
73/// - `idx_leaf`: which WOTS+ leaf inside that XMSS tree (low `h'` bits).
74pub fn ht_sign(
75    params: &HypertreeParams,
76    sk_seed: &[u8],
77    pk_seed: &[u8],
78    msg: &[u8],
79    mut idx_tree: u64,
80    mut idx_leaf: u32,
81) -> HypertreeSignature {
82    let h_prime = params.xmss.h_prime;
83    let mut sigs = Vec::with_capacity(params.d as usize);
84    let mut to_sign: Vec<u8> = msg.to_vec();
85
86    for j in 0..params.d {
87        let adrs = Adrs::new().with_xmss_height(j).with_xmss_index(idx_tree);
88        let sig_j = xmss_sign(&params.xmss, sk_seed, &to_sign, idx_leaf, pk_seed, adrs);
89
90        if j < params.d - 1 {
91            // Next layer signs the root recovered from this layer's signature.
92            to_sign = xmss_pk_from_sig(&params.xmss, idx_leaf, &sig_j, &to_sign, pk_seed, adrs);
93            advance_indices(&mut idx_tree, &mut idx_leaf, h_prime);
94        }
95        sigs.push(sig_j);
96    }
97    HypertreeSignature { xmss: sigs }
98}
99
100/// Verify a hypertree signature. `(idx_tree, idx_leaf)` identifies the layer-0
101/// leaf that was signed (same split as [`ht_sign`]).
102pub fn ht_verify(
103    params: &HypertreeParams,
104    pk_seed: &[u8],
105    pk_root: &[u8],
106    msg: &[u8],
107    sig: &HypertreeSignature,
108    mut idx_tree: u64,
109    mut idx_leaf: u32,
110) -> bool {
111    if sig.xmss.len() != params.d as usize {
112        return false;
113    }
114    let h_prime = params.xmss.h_prime;
115    let mut node: Vec<u8> = msg.to_vec();
116
117    for j in 0..params.d {
118        let adrs = Adrs::new().with_xmss_height(j).with_xmss_index(idx_tree);
119        node = xmss_pk_from_sig(
120            &params.xmss,
121            idx_leaf,
122            &sig.xmss[j as usize],
123            &node,
124            pk_seed,
125            adrs,
126        );
127        if j < params.d - 1 {
128            advance_indices(&mut idx_tree, &mut idx_leaf, h_prime);
129        }
130    }
131    node == pk_root
132}

134#[cfg(test)]
135mod tests {
136    use super::*;
137    use crate::sphincs::params::{
138        FipsParams, FIPS_128F, FIPS_128S, FIPS_192F, FIPS_192S, FIPS_256F, FIPS_256S,
139    };
140    use crate::sphincs::winternitz_ots::WotsParams;
141
142    /// End-to-end roundtrip: keygen → sign → verify, plus negative checks and
143    /// sig size formula `d · n · (len_wots + h')`.
144    fn run_e2e(p: &FipsParams) {
145        let params = HypertreeParams::new(
146            p.d,
147            XmssParams {
148                h_prime: p.h_prime,
149                wots: WotsParams::new(p.n, p.lg_w),
150            },
151        );
152        let n = p.n as usize;
153        let pk_seed: Vec<u8> = (0..n).map(|i| 0xA0u8.wrapping_add(i as u8)).collect();
154        let sk_seed: Vec<u8> = (0..n).map(|i| 0x50u8.wrapping_add(i as u8)).collect();
155        let msg: Vec<u8> = (0..n).map(|i| 0x10u8.wrapping_add(i as u8)).collect();
156
157        let pk_root = ht_pk_root(&params, &sk_seed, &pk_seed);
158
159        // Pick an idx somewhere below 2^h; split into (idx_tree, idx_leaf).
160        let global_idx: u64 = if params.h >= 64 {
161            u64::MAX / 3
162        } else {
163            ((1u64 << params.h) - 1) / 3
164        };
165        let idx_leaf = (global_idx & ((1u64 << p.h_prime) - 1)) as u32;
166        let idx_tree = global_idx >> p.h_prime;
167
168        let sig = ht_sign(&params, &sk_seed, &pk_seed, &msg, idx_tree, idx_leaf);
169        assert!(
170            ht_verify(&params, &pk_seed, &pk_root, &msg, &sig, idx_tree, idx_leaf),
171            "[{}] verify must accept good signature",
172            p.name,
173        );
174
175        let mut bad_msg = msg.clone();
176        bad_msg[0] ^= 1;
177        assert!(
178            !ht_verify(&params, &pk_seed, &pk_root, &bad_msg, &sig, idx_tree, idx_leaf),
179            "[{}] verify must reject tampered msg",
180            p.name,
181        );
182
183        let wrong_leaf = idx_leaf ^ 1;
184        assert!(
185            !ht_verify(&params, &pk_seed, &pk_root, &msg, &sig, idx_tree, wrong_leaf),
186            "[{}] verify must reject wrong idx_leaf",
187            p.name,
188        );
189
190        let num_elems = params.xmss.wots.num_elements() as usize;
191        let per_xmss = n * (num_elems + p.h_prime as usize);
192        assert_eq!(sig.size_in_bytes(), p.d as usize * per_xmss, "[{}]", p.name);
193    }
