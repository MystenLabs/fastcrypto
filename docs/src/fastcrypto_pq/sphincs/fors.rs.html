4//! FORS implementation from FIPS-205
5//!
6//! Every WOTS+ leaf at the bottom layer of the hypertree (layer=0) has a distinct FORS forest beneath it.
7//! So callers must set:
8//! - xmss height, xmss index & keypair address to that of the WOTS+ key
9//! - type to FORS_TREE
10
11use crate::sphincs::hash::tweakable_hash;
12use crate::sphincs::merkle::{build_auth_path, compute_root_from_path, merkle_node};
13use crate::sphincs::utils::{bits_to_base, bytes_to_bits};
14use crate::sphincs::{Adrs, AdrsType};
15
16// ------------------------------------------------------------------
17//                            Parameters
18// ------------------------------------------------------------------
19
20pub struct ForsParams {
21    pub k: u16, // number of trees in the FORS forest
22    pub a: u16, // height of each tree; also equivalent to #bits a single FORS tree can sign
23    pub t: u32, // t = 2^a
24    pub n: u16,
25}
26
27impl ForsParams {
28    pub fn new(k: u16, a: u16, n: u16) -> Self {
29        assert!(a < 32);
30        let t = 2u32.pow(a as u32);
31        ForsParams { k, a, t, n }
32    }
33}
34
35pub type ForsNode = Vec<u8>;
36
37pub type ForsSignature = Vec<ForsTreeSignature>; // k elements
38
39pub struct ForsTreeSignature {
40    pub sk: Vec<u8>,         // n bytes
41    pub auth: Vec<ForsNode>, // a elements
42}
43
44impl ForsTreeSignature {
45    pub fn size_in_bytes(&self) -> usize {
46        self.sk.len() + self.auth.iter().map(|n| n.len()).sum::<usize>()
47    }
48}

50// ------------------------------------------------------------------

54/// Build a FORS_TREE address at a given height/index. Assumes the caller's
55/// `adrs` is already a FORS_TREE address with `kp` set — true at every entry
56/// into this module (slh_dsa's `derive_fors_context` and the fors tests).
57fn fors_tree_adrs(adrs: Adrs, height: impl Into<u32>, index: u32) -> Adrs {
58    adrs.with_tree_height(height).with_tree_index(index)
59}
60
61fn fors_sk_gen(
62    params: &ForsParams,
63    sk_seed: &[u8],
64    pk_seed: &[u8],
65    adrs: Adrs,
66    idx: u32,
67) -> Vec<u8> {
68    assert!(idx < params.t * params.k as u32);
69
70    tweakable_hash(
71        pk_seed,
72        adrs.with_type(AdrsType::ForsPrf)
73            .with_key_pair_address(adrs.get_key_pair_address())
74            .with_tree_index(idx),
75        sk_seed,
76    )
77}
78
79/// F(sk) — the FORS leaf at forest-global index `idx`. Used by both the signer
80/// (with sk derived from sk_seed) and the verifier (with sk from the signature).
81fn fors_leaf(pk_seed: &[u8], adrs: Adrs, sk: &[u8], idx: u32) -> Vec<u8> {
82    tweakable_hash(pk_seed, fors_tree_adrs(adrs, 0u32, idx), sk)
83}
84
85fn fors_node(
86    params: &ForsParams,
87    sk_seed: &[u8],
88    i: u32,
89    z: u16,
90    pk_seed: &[u8],
91    adrs: Adrs,
92) -> ForsNode {
93    assert!(z <= params.a);
94    assert!(i < (params.k as u32) * 2u32.pow(u32::from(params.a - z)));
95
96    let leaf_generator = |idx| {
97        fors_leaf(
98            pk_seed,
99            adrs,
100            &fors_sk_gen(params, sk_seed, pk_seed, adrs, idx),
101            idx,
102        )
103    };
104    let adrs_generator = |height, idx| fors_tree_adrs(adrs, height, idx);
105    merkle_node(&leaf_generator, &adrs_generator, i, z, pk_seed)
106}

108/// Unpack `md` into `k` base-`a` digits per FIPS 205 Alg. 7. `md` is
109/// `⌈k·a / 8⌉` bytes; any trailing padding bits are discarded.
110fn md_to_indices(params: &ForsParams, md: &[u8]) -> Vec<u32> {
111    let ka = (params.k as usize) * (params.a as usize);
112    assert_eq!(md.len(), ka.div_ceil(8), "md must be ⌈k·a/8⌉ bytes");
113    let bits = bytes_to_bits(md);
114    let indices = bits_to_base(&bits[..ka], params.a);
115    debug_assert_eq!(indices.len(), params.k as usize);
116    indices
117}
118
119/// Compress the k FORS tree roots into the FORS public key (`T_k`).
120fn compress_roots(all_roots: &[u8], pk_seed: &[u8], adrs: Adrs) -> Vec<u8> {
121    tweakable_hash(
122        pk_seed,
123        adrs.with_type(AdrsType::ForsRoots)
124            .with_key_pair_address(adrs.get_key_pair_address()),
125        all_roots,
126    )
127}

129// ------------------------------------------------------------------

132
133pub fn fors_sign(
134    params: &ForsParams,
135    md: &[u8],
136    sk_seed: &[u8],
137    pk_seed: &[u8],
138    adrs: Adrs,
139) -> ForsSignature {
140    assert_eq!(
141        adrs.get_type(),
142        AdrsType::ForsTree,
143        "fors_sign expects a FORS_TREE ADRS"
144    );
145    let indices = md_to_indices(params, md);
146
147    let mut fors_sign = Vec::with_capacity(params.k as usize);
148    for i in 0..params.k as u32 {
149        let fors_idx = i * params.t + indices[i as usize];
150        let fors_tree_sk = fors_sk_gen(params, sk_seed, pk_seed, adrs, fors_idx);
151        // Indices are contiguous across the forest, so shift tree i's local sibling
152        // index into the global forest-wide index space at each height.
153        let node_at = |height, sib| {
154            let shift = i * 2u32.pow(u32::from(params.a - height));
155            fors_node(params, sk_seed, shift + sib, height, pk_seed, adrs)
156        };
157        let auth = build_auth_path(params.a, indices[i as usize], node_at);
158        fors_sign.push(ForsTreeSignature {
159            sk: fors_tree_sk,
160            auth,
161        });
162    }
163
164    fors_sign
165}

167pub fn fors_pk_from_sig(
168    params: &ForsParams,
169    sig: &ForsSignature,
170    md: &[u8],
171    pk_seed: &[u8],
172    adrs: Adrs,
173) -> Vec<u8> {
174    assert_eq!(
175        adrs.get_type(),
176        AdrsType::ForsTree,
177        "fors_pk_from_sig expects a FORS_TREE ADRS"
178    );
179    assert_eq!(sig.len(), params.k as usize);
180    let indices = md_to_indices(params, md);
181
182    let mut roots = Vec::with_capacity(params.k as usize);
183    let adrs_generator = |height, idx| fors_tree_adrs(adrs, height, idx);
184    for i in 0..params.k as u32 {
185        let tree_sig = &sig[i as usize];
186        assert_eq!(tree_sig.auth.len(), params.a as usize);
187
188        let leaf_idx = i * params.t + indices[i as usize];
189        let leaf = fors_leaf(pk_seed, adrs, &tree_sig.sk, leaf_idx);
190        let root = compute_root_from_path(leaf, leaf_idx, &tree_sig.auth, pk_seed, adrs_generator);
191        roots.push(root);
192    }
193
194    compress_roots(&roots.concat(), pk_seed, adrs)
195}

197#[cfg(test)]
198mod tests {
199    use super::*;
200    use crate::sphincs::params::{
201        FipsParams, FIPS_128F, FIPS_128S, FIPS_192F, FIPS_192S, FIPS_256F, FIPS_256S,
202    };
203
204    /// Compute the FORS public key from sk_seed by hashing the k tree roots — mirrors
205    /// what a signer would publish, independent of fors_sign/fors_pk_from_sig.
206    fn fors_pk_gen(params: &ForsParams, sk_seed: &[u8], pk_seed: &[u8], adrs: Adrs) -> Vec<u8> {
207        let roots: Vec<u8> = (0..params.k as u32)
208            .flat_map(|i| fors_node(params, sk_seed, i, params.a, pk_seed, adrs))
209            .collect();
210        compress_roots(&roots, pk_seed, adrs)
211    }
212
213    /// End-to-end roundtrip: keygen → sign → verify, plus a tampered-md negative
214    /// check and sig size formula `k · (1 + a) · n`.
215    fn run_e2e(p: &FipsParams) {
216        let params = ForsParams::new(p.k, p.a, p.n);
217        let n = p.n as usize;
218        let pk_seed: Vec<u8> = (0..n).map(|i| 0xA0u8.wrapping_add(i as u8)).collect();
219        let sk_seed: Vec<u8> = (0..n).map(|i| 0x50u8.wrapping_add(i as u8)).collect();
220        let adrs = Adrs::new()
221            .with_type(AdrsType::ForsTree)
222            .with_key_pair_address(7u32);
223
224        // md: ⌈k·a/8⌉ bytes of arbitrary content.
225        let md_len = ((p.k as usize) * (p.a as usize)).div_ceil(8);
226        let md: Vec<u8> = (0..md_len).map(|i| 0x5Au8.wrapping_add(i as u8)).collect();
227
228        let pk = fors_pk_gen(&params, &sk_seed, &pk_seed, adrs);
229        let sig = fors_sign(&params, &md, &sk_seed, &pk_seed, adrs);
230        let recovered = fors_pk_from_sig(&params, &sig, &md, &pk_seed, adrs);
231        assert_eq!(pk, recovered, "[{}] recovered pk must match", p.name);
232
233        let mut wrong = md;
234        wrong[0] ^= 1;
235        let wrong_pk = fors_pk_from_sig(&params, &sig, &wrong, &pk_seed, adrs);
236        assert_ne!(pk, wrong_pk, "[{}] wrong md must not recover pk", p.name);
237
238        let sig_size: usize = sig.iter().map(|t| t.size_in_bytes()).sum();
239        assert_eq!(
240            sig_size,
241            (p.k as usize) * (1 + p.a as usize) * n,
242            "[{}]",
243            p.name
244        );
245    }
