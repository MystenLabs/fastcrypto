4//! This module contains an implementation of the [ML-DSA-65](https://nvlpubs.nist.gov/nistpubs/FIPS/NIST.FIPS.204.pdf)
5//! signature scheme from FIPS 204, on top of the `mysten-mldsa-native-rs` wrapper around the
6//! mldsa-native C library.
7//!
8//! The private key is the 32-byte seed from which FIPS 204 derives the full signing key.
9//! Signing is hedged (the FIPS 204 default): each signature draws fresh randomness from the
10//! operating system, so two signatures over the same msg differ while both verify. The FIPS
11//! 204 context string is fixed to empty.
12//!
13//! Signatures are therefore neither reproducible nor unique for a given key and msg, so
14//! signature bytes must not be used as an identifier or a replay key.
15//!
16//! msgs can be signed and the signature can be verified again:
17//! ```rust
18//! # use fastcrypto_pq::mldsa65::*;
19//! # use fastcrypto::traits::{KeyPair, Signer, VerifyingKey};
20//! use rand::thread_rng;
21//! let keypair = MLDSA65KeyPair::generate(&mut thread_rng());
22//! let msg: &[u8] = b"Hello, Q world!";
23//! let signature = keypair.sign(msg);
24//! assert!(keypair.public().verify(msg, &signature).is_ok());
25//! ```

52/// The length of a private key in bytes: the FIPS 204 seed.
53pub const MLDSA65_PRIVATE_KEY_LENGTH: usize = mldsa::SEED_LENGTH;
54
55/// The length of a public key in bytes.
56pub const MLDSA65_PUBLIC_KEY_LENGTH: usize = mldsa::PUBLIC_KEY_LENGTH;
57
58/// The length of a signature in bytes.
59pub const MLDSA65_SIGNATURE_LENGTH: usize = mldsa::SIGNATURE_LENGTH;
60
61/// The key pair bytes length is the same as the private key length.
62pub const MLDSA65_KEYPAIR_LENGTH: usize = MLDSA65_PRIVATE_KEY_LENGTH;

64/// ML-DSA-65 public key.
65#[derive(Clone, PartialEq, Eq, Hash, PartialOrd, Ord)]
66pub struct MLDSA65PublicKey(mldsa::VerifyingKey);
67
68/// ML-DSA-65 private key: the 32-byte FIPS 204 seed, kept next to the expanded signing key
69/// derived from it so signing never re-runs key generation.
70///
71#[derive(SilentDebug, SilentDisplay)]
72pub struct MLDSA65PrivateKey {
73    seed: mldsa::SigningKeySeed,
74    signing_key: mldsa::SigningKey,
75}
76
77impl MLDSA65PrivateKey {
78    fn from_seed(seed: mldsa::SigningKeySeed) -> Self {
79        let (signing_key, _) = seed.expand();
80        MLDSA65PrivateKey { seed, signing_key }
81    }
82}
83
84impl PartialEq for MLDSA65PrivateKey {
85    fn eq(&self, other: &Self) -> bool {
86        self.seed == other.seed
87    }
88}
89
90impl Eq for MLDSA65PrivateKey {}
91
92impl ZeroizeOnDrop for MLDSA65PrivateKey {}

94/// ML-DSA-65 key pair.
95#[derive(Debug, PartialEq, Eq)]
96pub struct MLDSA65KeyPair {

101/// ML-DSA-65 signature.
102#[derive(Clone, PartialEq, Eq)]

134impl MLDSA65KeyPair {
135    /// Build the pair with a single key expansion
136    fn from_seed(seed: mldsa::SigningKeySeed) -> Self {
137        let (signing_key, verifying_key) = seed.expand();
138        MLDSA65KeyPair {
139            public: MLDSA65PublicKey(verifying_key),
140            private: MLDSA65PrivateKey { seed, signing_key },
141        }
142    }
143}
144
145/// Recovers the public key from a standalone private key by re-expanding its seed.
146/// Other construction paths use [`MLDSA65KeyPair::from_seed`] to expand once.
147impl From<MLDSA65PrivateKey> for MLDSA65KeyPair {
148    fn from(private: MLDSA65PrivateKey) -> Self {
149        let public = MLDSA65PublicKey::from(&private);
150        MLDSA65KeyPair { public, private }
151    }
152}

154/// The bytes form of the keypair always only contain the private key bytes

158        Ok(MLDSA65KeyPair::from_seed(seed))

170impl KeyPair for MLDSA65KeyPair {
171    type PubKey = MLDSA65PublicKey;
172    type PrivKey = MLDSA65PrivateKey;
173    type Sig = MLDSA65Signature;
174
175    fn public(&'_ self) -> &'_ Self::PubKey {
176        &self.public
177    }
178
179    fn private(self) -> Self::PrivKey {
180        self.private
181    }
182
183    // Not cfg-gated on copy_key; this crate always enables fastcrypto/copy_key
184    fn copy(&self) -> Self {
185        MLDSA65KeyPair {
186            public: self.public.clone(),
187            private: MLDSA65PrivateKey::from_seed(
188                mldsa::SigningKeySeed::from_bytes(self.private.seed.as_ref())
189                    .expect("a seed has the seed length"),
190            ),
191        }
192    }
193
194    fn generate<R: AllowedRng>(rng: &mut R) -> Self {
195        let mut seed = Zeroizing::new([0u8; MLDSA65_PRIVATE_KEY_LENGTH]);
196        rng.fill_bytes(seed.as_mut());
197        MLDSA65KeyPair::from_seed(mldsa::SigningKeySeed::from(*seed))
198    }
199}

201impl FromStr for MLDSA65KeyPair {
202    type Err = FastCryptoError;

204    fn from_str(s: &str) -> Result<Self, Self::Err> {
205        Self::decode_base64(s)
206    }
207}

209impl Signer<MLDSA65Signature> for MLDSA65KeyPair {
210    fn sign(&self, msg: &[u8]) -> MLDSA65Signature {
211        // Hedged signing, the FIPS 204 default: fresh randomness from the operating system
212        // (OsRng/getrandom) for every sig.
213        let mut rnd = [0u8; mldsa::RND_LENGTH];
214        OsRng.fill_bytes(&mut rnd);
215        MLDSA65Signature(
216            self.private
217                .signing_key
218                .sign(msg, b"", &rnd)
219                .expect("the empty context cannot exceed the length limit"),
220        )
221    }
222}

268impl<'a> From<&'a MLDSA65PrivateKey> for MLDSA65PublicKey {
269    fn from(private: &'a MLDSA65PrivateKey) -> Self {
270        // re-expands the seed; callers that need the public key repeatedly
271        // should hold a MLDSA65KeyPair, which caches it
272        let (_, public) = private.seed.expand();
273        MLDSA65PublicKey(public)
274    }
275}
276
277impl ToFromBytes for MLDSA65PublicKey {
278    fn from_bytes(bytes: &[u8]) -> Result<Self, FastCryptoError> {
279        mldsa::VerifyingKey::from_bytes(bytes)
280            .map(MLDSA65PublicKey)
281            .map_err(|_| InvalidInput)
282    }
283}

312impl VerifyingKey for MLDSA65PublicKey {
313    type PrivKey = MLDSA65PrivateKey;
314    type Sig = MLDSA65Signature;
315    const LENGTH: usize = MLDSA65_PUBLIC_KEY_LENGTH;
316
317    fn verify(&self, msg: &[u8], signature: &MLDSA65Signature) -> Result<(), FastCryptoError> {
318        // Every failure maps to InvalidSignature without classification
319        self.0
320            .verify(msg, b"", &signature.0)
321            .map_err(|_| InvalidSignature)
322    }
