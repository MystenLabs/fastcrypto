4//! SLIP-0010 key derivation for the post-quantum schemes.
5
6use fastcrypto::error::FastCryptoResult;
7use fastcrypto::slip10::{derive_hardened, Slip10MasterKey};
8use fastcrypto::traits::ToFromBytes;
9
10use crate::mldsa65::MLDSA65KeyPair;
11
12/// Derive an ML-DSA-65 keypair from a BIP-39 seed along a hardened SLIP-0010
13/// path, per [satoshilabs/slips#1968](https://github.com/satoshilabs/slips/pull/1968):
14/// the node secret is the FIPS 204 keygen seed.
15pub fn derive_mldsa65_keypair(seed: &[u8], indexes: &[u32]) -> FastCryptoResult<MLDSA65KeyPair> {
16    let node = derive_hardened(Slip10MasterKey::MlDsa65, seed, indexes)?;
17    MLDSA65KeyPair::from_bytes(&node.secret)
18}
