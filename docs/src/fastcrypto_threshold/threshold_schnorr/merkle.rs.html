4//! A [NestedMerkleTree] commits to a sequence of rows of leaves. Each row is committed by its own
5//! Merkle tree (its `row_root`). A top tree then commits to those row roots. A
6//! [NestedMerkleProof] for a leaf carries both an inclusion proof against its row root and an
7//! inclusion proof binding that row root to the top root.
8
9use fastcrypto::error::FastCryptoError::{InvalidInput, InvalidMessage, InvalidProof};
10use fastcrypto::error::FastCryptoResult;
11use fastcrypto::hash::Blake2b256;
12use fastcrypto::merkle::{MerkleProof, MerkleTree, Node};
13use serde::{Deserialize, Serialize};
14
15/// A two-level Merkle commitment to a sequence of rows of leaves.
16pub struct NestedMerkleTree {
17    row_trees: Vec<MerkleTree<Blake2b256>>,
18    top_tree: MerkleTree<Blake2b256>,
19}
20
21/// Inclusion proof for a leaf in a [NestedMerkleTree]: row proof up to the (implied) row root,
22/// then top proof binding that row root to the dispersal's top root. The intermediate row root
23/// is derived from `row_proof` + leaf during verification — not stored.
24#[derive(Clone, Debug, Serialize, Deserialize)]
25pub struct NestedMerkleProof {
26    pub row_proof: MerkleProof,
27    pub top_proof: MerkleProof,
28}
29
30#[allow(dead_code)]
31impl NestedMerkleTree {
32    /// Build a tree committing to `rows`. `rows[i][j]` is the `j`-th leaf in row `i`.
33    pub fn new<S: Serialize>(rows: impl IntoIterator<Item = Vec<S>>) -> FastCryptoResult<Self> {
34        let row_trees: Vec<MerkleTree<Blake2b256>> = rows
35            .into_iter()
36            .map(|row| MerkleTree::<Blake2b256>::build_from_unserialized(row.iter()))
37            .collect::<FastCryptoResult<_>>()?;
38        let row_roots: Vec<Node> = row_trees.iter().map(|t| t.root()).collect();
39        let top_tree = MerkleTree::<Blake2b256>::build_from_unserialized(row_roots.iter())?;
40        Ok(Self {
41            row_trees,
42            top_tree,
43        })
44    }
45
46    /// The top tree's root — the overall commitment.
47    pub fn top_root(&self) -> Node {
48        self.top_tree.root()
49    }
50
51    /// Root of the row tree for row `row_idx`, or `None` if out of bounds.
52    pub fn row_root(&self, row_idx: usize) -> Option<Node> {
53        self.row_trees.get(row_idx).map(|t| t.root())
54    }
55
56    /// Inclusion proof for the leaf at position `leaf_idx` in row `row_idx`.
57    pub fn get_proof(
58        &self,
59        row_idx: usize,
60        leaf_idx: usize,
61    ) -> FastCryptoResult<NestedMerkleProof> {
62        let row_tree = self.row_trees.get(row_idx).ok_or(InvalidInput)?;
63        Ok(NestedMerkleProof {
64            row_proof: row_tree.get_proof(leaf_idx)?,
65            top_proof: self.top_tree.get_proof(row_idx)?,
66        })
67    }
68
69    /// Compute the root of a row's Merkle subtree, byte-identical to what [Self::new] would build
70    /// for the same row.
71    pub fn compute_row_root<S: Serialize>(row: &[S]) -> FastCryptoResult<Node> {
72        Ok(MerkleTree::<Blake2b256>::build_from_unserialized(row.iter())?.root())
73    }
74}
75
76impl NestedMerkleProof {
77    /// Verify this proof against `top_root` for `leaf` at position (`row_idx`, `leaf_idx`) of a
78    /// tree with `n_rows` rows of `n_leaves` leaves each.
79    pub fn verify<S: Serialize>(
80        &self,
81        top_root: &Node,
82        leaf: &S,
83        row_idx: usize,
84        leaf_idx: usize,
85        n_rows: usize,
86        n_leaves: usize,
87    ) -> FastCryptoResult<()> {
88        (self.derive_top_root(leaf, row_idx, leaf_idx, n_rows, n_leaves)? == *top_root)
89            .then_some(())
90            .ok_or(InvalidProof)
91    }
92
93    // TODO: Remove this and `derive_row_root_for_trusted_tree` once AVID complaints are checked
94    // with the strict variants.
95    /// Verify this proof against `top_root` for `leaf` at position (`row_idx`, `leaf_idx`),
96    /// assuming a trusted tree, i.e. one built by [NestedMerkleTree]. If the tree cannot be
97    /// trusted, use [Self::verify] instead.
98    pub fn verify_for_trusted_tree<S: Serialize>(
99        &self,
100        top_root: &Node,
101        leaf: &S,
102        row_idx: usize,
103        leaf_idx: usize,
104    ) -> FastCryptoResult<()> {
105        let row_root = self.derive_row_root_for_trusted_tree(leaf, leaf_idx)?;
106        let derived_top_root = self
107            .top_proof
108            .compute_root_for_trusted_tree(
109                &bcs::to_bytes(&row_root).map_err(|_| InvalidInput)?,
110                row_idx,
111            )
112            .ok_or(InvalidMessage)?;
113        (derived_top_root == *top_root)
114            .then_some(())
115            .ok_or(InvalidProof)
116    }
117
118    /// Derive the implied row root from this proof's row portion and `leaf` at `leaf_idx`,
119    /// assuming a trusted tree, i.e. one built by [NestedMerkleTree].
120    /// Returns [InvalidMessage] if the path is malformed.
121    pub fn derive_row_root_for_trusted_tree<S: Serialize>(
122        &self,
123        leaf: &S,
124        leaf_idx: usize,
125    ) -> FastCryptoResult<Node> {
126        self.row_proof
127            .compute_root_for_trusted_tree(
128                &bcs::to_bytes(leaf).map_err(|_| InvalidInput)?,
129                leaf_idx,
130            )
131            .ok_or(InvalidMessage)
132    }
133
134    /// Derive the implied top root from this proof and `leaf` at (`row_idx`, `leaf_idx`) of a tree
135    /// with `n_rows` rows of `n_leaves` leaves each: walk the row proof from the leaf to an
136    /// implied row root, then walk the top proof from that row root to an implied top root.
137    /// Returns [InvalidMessage] if either path is malformed or has the wrong length, or if either
138    /// index is outside the tree.
139    pub fn derive_top_root<S: Serialize>(
140        &self,
141        leaf: &S,
142        row_idx: usize,
143        leaf_idx: usize,
144        n_rows: usize,
145        n_leaves: usize,
146    ) -> FastCryptoResult<Node> {
147        let row_root = self
148            .row_proof
149            .compute_root(
150                &bcs::to_bytes(leaf).map_err(|_| InvalidInput)?,
151                leaf_idx,
152                n_leaves,
153            )
154            .ok_or(InvalidMessage)?;
155        self.top_proof
156            .compute_root(
157                &bcs::to_bytes(&row_root).map_err(|_| InvalidInput)?,
158                row_idx,
159                n_rows,
160            )
161            .ok_or(InvalidMessage)
162    }
163}
