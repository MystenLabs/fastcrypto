4//! Knapsack-verified weight reduction.
5//!
6//! Replaces weights `w_1..w_n` (total `W`) by smaller weights `w'_1..w'_n`
7//! (total `W'`) and thresholds `t', f'`, minimizing `W'` while preserving the
8//! security of the weighted protocols when instantiated with
9//! `(t', f', W')` in place of `(t, f, W)`. Inputs are the privacy threshold
10//! `t`, the Byzantine weight bound `f` (with `0 < f < t`, `t + 2f <= W` and
11//! `t + f + delta <= W` enforced), and the
12//! allowed liveness degradation `delta` in original weight units.
13//!
14//! # Guarantees
15//!
16//! Let `g(b) = max { w'(S) : w(S) <= b }`, computed exactly by a knapsack DP,
17//! and set `t' = g(t-1) + 1`, `f' = g(f)`. For every subset S of parties:
18//!
19//! - (P) Privacy: `w(S) < t  =>  w'(S) < t'`.
20//! - (L1) Liveness: `w(S) >= t + f + delta  =>  w'(S) >= t' + f'`. Sets that
21//!   could form a Byzantine-tolerant quorum originally still can.
22//! - (L2) Byzantine bound: `w(S) <= f  =>  w'(S) <= f'`. Limits how much of
23//!   the (L1) quorum might be Byzantine.
24//! - (L3) Liveness: `w(S) >= t + delta  =>  w'(S) >= t'`. Sets that could
25//!   reconstruct originally (with `delta` slack) still reach the reduced
26//!   threshold.
27//!
28//! `t > f` => `t' > f'` follows from monotonicity of `g`, and the output
29//! satisfies `t' > f' >= 1` (candidates with `f' == 0` are rejected).
30//!
31//! # Algorithm
32//!
33//! Candidates are scaled roundings `w'_i = floor(w_i / d + c)` for divisors
34//! `d` on a 0.01 grid and rounding offsets `c in [0, 1)` on a 0.1 grid.
35//! For a fixed offset, `W'` is non-increasing in `d`, so the sweep goes downward
36//! and returns the first candidate that passes the feasibility check; the best
37//! result over all offsets is returned. Each sweep starts at the largest useful
38//! divisor (`max weight / (1 - c)`, above which all reduced weights are zero).
39//! Consecutive grid points usually produce identical reduced vectors, so the
40//! sweep jumps directly between "breakpoints" where some `w'_i` changes.
41//!
42//! The same DP powers `verify_reduction`, an `O(n * W')` exact checker of the
43//! above four properties for an arbitrary candidate `(w', t', f')`.
44//!
45//! The implementation follows the weight-reduction framework of Swiper (Tonkikh and
46//! Freitas, PODC 2024, <https://arxiv.org/abs/2307.15561>): scaled-rounding
47//! candidates verified via knapsack computations.
48//! Our variant differs in enforcing the four absolute predicates above with tight
49//! output thresholds `t', f'`, trying a grid of rounding offsets, and sweeping
50//! divisors downward instead of binary search, as our feasibility predicate is
51//! non-monotone in `d`.
52
53use fastcrypto::error::{FastCryptoError, FastCryptoResult};
54
55const MAX_PARTIES: usize = 1000;
56const MAX_WEIGHT: u16 = 10_000;
57
58#[derive(Clone, Debug, PartialEq, Eq)]
59pub struct ReducedWeights {
60    /// Same order as the input, entries may be zero.
61    pub weights: Vec<u16>,
62    pub t: u16,
63    pub f: u16,
64}
65
66pub(crate) fn reduce_weights(
67    weights: &[u16],
68    t: u16,
69    f: u16,
70    delta: u16,
71    total_weight_lower_bound: u16,
72) -> FastCryptoResult<ReducedWeights> {
73    // Sanity checks on input parameters.
74    if weights.is_empty() || weights.len() > MAX_PARTIES || weights.iter().any(|&w| w > MAX_WEIGHT)
75    {
76        return Err(FastCryptoError::InvalidInput);
77    }
78    let total_weight = weights.iter().map(|&w| w as u32).sum::<u32>();
79    if total_weight == 0 || total_weight > u16::MAX as u32 {
80        return Err(FastCryptoError::InvalidInput);
81    }
82    let w_total = total_weight as u16;
83    if t == 0
84        || f == 0
85        || t > w_total
86        || f >= t
87        || (t as u32) + 2 * (f as u32) > total_weight
88        // While in practice we would only use delta < f, what is needed
89        // for liveness is t + f + delta <= W
90        || (t as u32) + (f as u32) + (delta as u32) > total_weight
91    {
92        return Err(FastCryptoError::InvalidInput);
93    }
94    if total_weight_lower_bound == 0 || total_weight_lower_bound > w_total {
95        return Err(FastCryptoError::InvalidInput);
96    }
97
98    // Prefer fewer total shares, then a lower reconstruction threshold.
99    let rank = |c: &ReducedWeights| (c.weights.iter().map(|&w| w as u32).sum::<u32>(), c.t);
100
101    // Start from the identity fallback (d = 1).
102    let mut best = ReducedWeights {
103        weights: weights.to_vec(),
104        t,
105        f,
106    };
107    // Try all rounding offsets on the 0.1 grid and keep the best.
108    for offset in (0..100).step_by(10) {
109        let best_total = rank(&best).0;
110        if let Some(candidate) = sweep(
111            weights,
112            w_total,
113            t,
114            f,
115            delta,
116            total_weight_lower_bound,
117            offset,
118            best_total,
119        ) {
120            if rank(&candidate) < rank(&best) {
121                best = candidate;
122            }
123        }
124    }
125    Ok(best)
126}

128/// Sweep divisors downward (largest first) on the 0.01 grid for one rounding
129/// offset, jumping between "breakpoints" where the reduced vector changes, and return the first candidate that passes the
130/// feasibility check. Since `W'` only grows as `d` shrinks, the sweep gives up
131/// once the candidate total exceeds `best_total`.
132#[allow(clippy::too_many_arguments)]
133fn sweep(
134    weights: &[u16],
135    w_total: u16,
136    t: u16,
137    f: u16,
138    delta: u16,
139    lower_bound: u16,
140    offset: u64,
141    best_total: u32,
142) -> Option<ReducedWeights> {
143    let max_weight = *weights.iter().max().expect("non-empty") as u64;
144
145    // The candidate divisor currently under test, times 100 (0.01 grid),
146    // starting from the largest useful divisor: to have at least one party
147    // with non-zero reduced weight, we need floor(w/d + c) > 0
148    // <-> w/d >= 1 - c <-> d <= w/(1-c), i.e., d <= 10000w/(100-offset) on the
149    // grid.
150    let mut d_candidate = (10_000 * max_weight / (100 - offset)) as u32;
151
152    while d_candidate > 100 {
153        let reduced = weights
154            .iter()
155            .map(|&w| reduce_weight(w, d_candidate, offset))
156            .collect::<Vec<_>>();
157        let reduced_total = reduced.iter().map(|&w| w as u32).sum::<u32>();
158        if reduced_total > best_total {
159            return None;
160        }
161        if reduced_total >= lower_bound as u32
162            && !greedy_reject(weights, &reduced, w_total, t, f, delta, reduced_total)
163        {
164            if let Ok((tp, fp)) = check_candidate(weights, w_total, t, f, delta, &reduced) {
165                return Some(ReducedWeights {
166                    weights: reduced,
167                    t: tp,
168                    f: fp,
169                });
170            }
171        }
172
173        // Jump to the largest divisor where some reduced weight increases; all
174        // divisors in between yield the current (just rejected) vector again.
175        // Reduced weights only grow as d shrinks, and a party at q reaches q+1
176        // exactly when floor(w/d' + c) >= q+1 <-> w/d' >= q+1-c
177        // <-> d' <= w/(q+1-c), i.e., d' <= 10000w/(100(q+1) - offset) on the
178        // grid.
179        let next = weights
180            .iter()
181            .zip(reduced.iter())
182            .map(|(&w, &q)| (10_000 * (w as u64) / (100 * (q as u64 + 1) - offset)) as u32)
183            .max()
184            .expect("non-empty");
185        d_candidate = next.min(d_candidate - 1);
186    }
187    None
188}

190/// `floor(w / (d/100) + offset/100) = floor((10000w + offset * d) / (100d))`.
191fn reduce_weight(w: u16, d: u32, offset: u64) -> u16 {
192    ((10_000 * (w as u64) + offset * (d as u64)) / (100 * (d as u64))) as u16
193}

195/// The value-space knapsack table `min_original_weight[v] = min { w(S) : w'(S) = v }` for
196/// `v in [0, W']`, computed in `O(n * W')`.
197fn knapsack_min_original(weights: &[u16], reduced: &[u16], reduced_total: u32) -> Vec<u32> {
198    let mut min_original_weight = vec![u32::MAX; reduced_total as usize + 1];
199    min_original_weight[0] = 0;
200    for (&w, &q) in weights.iter().zip(reduced.iter()) {
201        let (w, q) = (w as u32, q as usize);
202        if q == 0 {
203            continue;
204        }
205        for v in (q..min_original_weight.len()).rev() {
206            if min_original_weight[v - q] != u32::MAX {
207                min_original_weight[v] = min_original_weight[v].min(min_original_weight[v - q] + w);
208            }
209        }
210    }
211    min_original_weight
212}

214/// `g(weight) = max { w'(S) : w(S) <= weight }`.
215/// In other words, the largest reduced weight a subset can have while its
216/// original weight <= `weight`: w(S) <= weight => w'(S) <= g(weight).
217fn max_reduced_weight(min_original_weight: &[u32], weight: u32) -> u32 {
218    min_original_weight
219        .iter()
220        .rposition(|&x| x <= weight)
221        .expect("min_original_weight[0] = 0 always qualifies") as u32
222}

224/// A cheap rejection test that costs `O(n log n)`. It never errs when claiming
225/// infeasibility.
226fn greedy_reject(
227    weights: &[u16],
228    reduced: &[u16],
229    w_total: u16,
230    t: u16,
231    f: u16,
232    delta: u16,
233    reduced_total: u32,
234) -> bool {
235    // Sort by "density", i.e., reduced_weight/original_weight, densest first.
236    // Densities are compared using cross-multiplication,
237    // reduced_weight_a / original_weight_a > reduced_weight_b / original_weight_b <=>
238    // reduced_weight_a * original_weight_b > reduced_weight_b * original_weight_a
239    let mut by_density = (0..weights.len())
240        .filter(|&i| reduced[i] > 0)
241        .collect::<Vec<_>>();
242    by_density.sort_by(|&a, &b| {
243        ((reduced[b] as u32) * (weights[a] as u32))
244            .cmp(&((reduced[a] as u32) * (weights[b] as u32)))
245    });
246
247    // Lower bound on g(weight): *any* ordering of the parties would work for a lower bound.
248    // We chose the density-sorted ordering because it can be proven to guarantee:
249    //  g(weight) − g_lower(weight) < max_i reduced_weight
250    // thus fairly tight.
251    let g_lower = |weight: u32| {
252        let (mut remaining_weight, mut reduced_weight) = (weight, 0u32);
253        for &i in &by_density {
254            let w = weights[i] as u32;
255            if w <= remaining_weight {
256                remaining_weight -= w;
257                reduced_weight += reduced[i] as u32;
258            }
259        }
260        reduced_weight
261    };
262
263    // 1. (L3) requires w(S) >= t + delta => w'(S) >= t'. A set S has
264    //    w(S) >= t + delta exactly when its complement T fits the budget
265    //    w(T) = W - w(S) <= W - t - delta, and w'(S) = W' - w'(T), so
266    //    minimizing w'(S) is maximizing w'(T):
267    //      min { w'(S) : w(S) >= t + delta } = W' - g(W-t-delta) >= t'.
268    // 2. Substituting t' = g(t-1) + 1 and rearranging:
269    //      (L3) => g(t-1) + g(W-t-delta) <= W' - 1.
270    //    Doing the same for (L1) with t' + f' = g(t-1) + 1 + g(f):
271    //      (L1) => g(t-1) + g(f) + g(W-t-f-delta) <= W' - 1.
272    // 3. g_lower(b) <= g(b) by construction.
273
274    // Lower bound on (L3): g_lower(t-1) + g_lower(W-t-delta) <= W' - 1.
275    if g_lower((t - 1) as u32) + g_lower(w_total as u32 - t as u32 - delta as u32) >= reduced_total
276    {
277        return true;
278    }
279    // Lower bound on (L1): g_lower(t-1) + g_lower(f) + g_lower(W-t-f-delta) <= W' - 1.
280    g_lower((t - 1) as u32)
281        + g_lower(f as u32)
282        + g_lower(w_total as u32 - t as u32 - f as u32 - delta as u32)
283        >= reduced_total
284}

286/// Feasibility check of a candidate. Returns `(t', f')` if all constraints hold,
287/// an error if liveness fails.
288fn check_candidate(
289    weights: &[u16],
290    w_total: u16,
291    t: u16,
292    f: u16,
293    delta: u16,
294    reduced: &[u16],
295) -> FastCryptoResult<(u16, u16)> {
296    let reduced_total = reduced.iter().map(|&w| w as u32).sum::<u32>();
297    let min_original_weight_map = knapsack_min_original(weights, reduced, reduced_total);
298
299    // t' is the smallest threshold that maintains privacy (P): w(S) < t => w'(S) < t'.
300    let tp = max_reduced_weight(&min_original_weight_map, (t - 1) as u32) + 1;
301
302    // f' is the smallest bound satisfying (L2): w(S) <= f => w'(S) <= f'.
303    let fp = max_reduced_weight(&min_original_weight_map, f as u32);
304    // Downstream protocols require positive thresholds.
305    if fp == 0 {
306        return Err(FastCryptoError::GeneralError("f' == 0".to_string()));
307    }
308
309    // Liveness (L3): w(S) >= t + delta => w'(S) >= t'.
310    // Checked via the complement T of the worst such S:
311    //   min { w'(S) : w(S) >= t + delta } = W' - max { w'(T) : w(T) <= W - t - delta }.
312    let b1 = (w_total as u32) - (t as u32 + delta as u32);
313    if reduced_total - max_reduced_weight(&min_original_weight_map, b1) < tp {
314        return Err(FastCryptoError::GeneralError("L3 violated".to_string()));
315    }
316
317    // Liveness (L1): w(S) >= t + f + delta => w'(S) >= t' + f'.
318    // Checked via the complement T of the worst such S:
319    //   min { w'(S) : w(S) >= t + f + delta } = W' - max { w'(T) : w(T) <= W - t - f - delta }.
320    let b2 = (w_total as u32) - (t as u32 + f as u32 + delta as u32);
321    if reduced_total - max_reduced_weight(&min_original_weight_map, b2) < tp + fp {
322        return Err(FastCryptoError::GeneralError("L1 violated".to_string()));
323    }
324
325    // tp <= t and fp <= f, so both fit u16.
326    Ok((tp as u16, fp as u16))
327}

329/// Verifier for a candidate reduction: checks conditions (P), (L1), (L2), and (L3)
330/// for an arbitrary `(weights', t', f')` in `O(n * W')` time. Returns
331/// `InvalidInput` for malformed inputs and a `GeneralError` naming the first
332/// violated condition otherwise.
333pub(crate) fn verify_reduction(
334    weights: &[u16],
335    t: u16,
336    f: u16,
337    delta: u16,
338    reduction: &ReducedWeights,
339) -> FastCryptoResult<()> {
340    let w_total = weights.iter().map(|&w| w as u32).sum::<u32>();
341    let reduced_total = reduction.weights.iter().map(|&w| w as u32).sum::<u32>();
342    if weights.len() != reduction.weights.len()
343        || weights.len() > MAX_PARTIES
344        || t == 0
345        || f == 0
346        || (t as u32) > w_total
347        || w_total > u16::MAX as u32
348        || reduced_total > u16::MAX as u32
349        || reduction.t == 0
350        || reduction.f == 0
351    {
352        return Err(FastCryptoError::InvalidInput);
353    }
354    // Same sanity checks as reduce_weights, in both spaces.
355    if f >= t
356        || reduction.f >= reduction.t
357        || (t as u32) + 2 * (f as u32) > w_total
358        || (t as u32) + (f as u32) + (delta as u32) > w_total
359    {
360        return Err(FastCryptoError::InvalidInput);
361    }
362    let min_original_weight_map = knapsack_min_original(weights, &reduction.weights, reduced_total);
363
364    // (P): max { w'(S) : w(S) <= t-1 } <= t' - 1.
365    if max_reduced_weight(&min_original_weight_map, (t - 1) as u32) > (reduction.t - 1) as u32 {
366        return Err(FastCryptoError::GeneralError("P violated".to_string()));
367    }
368
369    // (L2): max { w'(S) : w(S) <= f } <= f'.
370    if max_reduced_weight(&min_original_weight_map, f as u32) > reduction.f as u32 {
371        return Err(FastCryptoError::GeneralError("L2 violated".to_string()));
372    }
373
374    // (L3): min { w'(S) : w(S) >= t + delta } >= t'.
375    let b1 = w_total - (t as u32 + delta as u32);
376    if reduced_total - max_reduced_weight(&min_original_weight_map, b1) < reduction.t as u32 {
377        return Err(FastCryptoError::GeneralError("L3 violated".to_string()));
378    }
379
380    // (L1): min { w'(S) : w(S) >= t + f + delta } >= t' + f'.
381    let b2 = w_total - (t as u32 + f as u32 + delta as u32);
382    if reduced_total - max_reduced_weight(&min_original_weight_map, b2)
383        < reduction.t as u32 + reduction.f as u32
384    {
385        return Err(FastCryptoError::GeneralError("L1 violated".to_string()));
386    }
387    Ok(())
