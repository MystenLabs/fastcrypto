4//! This module implements threshold Schnorr signatures.
5//! The signatures are compatible with BIP-0340.
6//!
7//! It provides the following protocols:
8//!
9//! 1. A Distributed Key Generation (DKG) protocol to generate a shared signing key without a
10//!    trusted dealer. The protocol also allows resharing of a share from a previous DKG, allowing
11//!    for key rotation. This is implemented in the [avss] module.
12//! 2. A protocol to generate a batch of secret shared nonces for signing. This is implemented in
13//!    the [batch_avss_avid] module.
14//! 3. A presigning protocol to create presigning tuples from the secret shared nonces. This is
15//!    implemented in the [presigning] module. The presigning tuples can be created in advance of
16//!    knowing the message to be signed, and one tuple is consumed for each signature.
17//! 4. A signing protocol which allows parties to create partial signatures from a presigning
18//!    tuple and aggregate them into a full signature if there are enough partial signatures. This
19//!    is implemented in the [signing] module.
20//!
21//! For both the DKG and nonce generation protocols, it is assumed that each party has an
22//! encryption key pair (ECIES) and these public keys are known to all parties. These can be
23//! reused for all instances of the protocols.
24//!
25//! It is also assumed that all messages between parties are sent over authenticated channels, so
26//! that the receiver of a message knows who sent it and that it was not modified.
27//!
28//! The thresholds are defined as follows:
29//! * <i>W</i> = total weight of all parties
30//! * <i>f</i> = maximum Byzantine weight
31//! * <i>t</i> = threshold for signing
32//!
33//! For the weights used here, [Parameters::validate] checks the basic invariants `t < W`,
34//! `t &geq; f` and `t + f &leq; W`. The AVID-based nonce protocol additionally requires `W > 2f`
35//! (enforced in `Avid::new`).
36

74/// Threshold parameters for the AVSS protocols.
75#[derive(Copy, Clone, Debug)]
76pub struct Parameters {
77    /// Reconstruction threshold: `≥ t` valid shares (by weight) reconstruct a secret.
78    pub t: u16,
79    /// Byzantine bound by share-weight.
80    pub f: u16,
81}
82
83impl Parameters {
84    /// Validate `(t, f)` against the given total weight `W`, checking the basic invariants needed
85    /// for the sharing here: `0 < f`, `t < W`, `t ≥ f` and `t + f ≤ W`. Note the AVID-based nonce
86    /// protocol has a further requirement, `W > 2f`, which is enforced when its Reed-Solomon coder
87    /// is built (`Avid::new`), not here.
88    pub fn validate(&self, total_weight: u16) -> FastCryptoResult<()> {
89        let Parameters { t, f } = *self;
90        if f == 0
91            || t == 0
92            || t >= total_weight
93            || t < f
94            || t as u32 + f as u32 > total_weight as u32
95        {
96            return Err(InvalidInput);
97        }
98        Ok(())
99    }
100}

114/// This represents a certificate over a payload that a subset of the parties have signed.
115/// Here, the implementation is abstract, and it is up to the caller to implement the actual
116/// verification functionality.
117pub trait Certificate {
118    type Payload;
119
120    fn signers(&self) -> &BTreeSet<PartyId>;
121
122    fn payload(&self) -> &Self::Payload;
123
124    fn verify(&self) -> FastCryptoResult<()>;
125
126    fn to_verified(&self) -> FastCryptoResult<VerifiedCertificate<Self>>
127    where
128        Self: Clone,
129    {
130        self.verify().map(|_| VerifiedCertificate(self.clone()))
131    }
132}

134/// A [Certificate] that has already been verified.
135pub struct VerifiedCertificate<C>(C);
136
137impl<C: Certificate> VerifiedCertificate<C> {
138    pub fn certificate(&self) -> &C {
139        &self.0
140    }
141
142    pub fn payload(&self) -> &C::Payload {
143        self.0.payload()
144    }
145}

593    /// Restrict a `HashMap` to a given set of keys.
594    /// Panics if the given subset is not a subset of the maps' keys.
595    pub(crate) fn restrict<T: Clone, I: Eq + Hash>(
596        map: &HashMap<I, T>,
597        keys_subset: impl Iterator<Item = I>,
598    ) -> HashMap<I, T>
599    where
600        usize: From<I>,
601    {
602        keys_subset
603            .map(|i| {
604                let value = map.get(&i).unwrap().clone();
605                (i, value)
606            })
607            .collect()
608    }

729    /// Sign with every combination of the Y parities of the verifying key, the nonce R and the
730    /// derived verifying key, since each selects a different branch in the BIP-0340 adjustments.
731    #[test]
732    fn test_signing_all_parities() {
733        let (t, n) = (3u16, 5u16);
734        let message = b"parity";
735        let mut rng = rand::thread_rng();
736        let has_even_y = |p: &G| p.has_even_y().unwrap();
737
738        for vk_even in [true, false] {
739            let sk = loop {
740                let sk = S::rand(&mut rng);
741                if has_even_y(&(G::generator() * sk)) == vk_even {
742                    break sk;
743                }
744            };
745            let vk = G::generator() * sk;
746            let sk_shares = mock_shares(&mut rng, sk, t, n);
747
748            // No derivation, and addresses whose derived verifying keys have even and odd Y.
749            let address_with = |derived_even: bool| -> Address {
750                (0u8..=255)
751                    .map(|i| [i; 32])
752                    .find(|a| {
753                        has_even_y(&derive_verifying_key_internal(&vk, a).unwrap()) == derived_even
754                    })
755                    .unwrap()
756            };
757            for address in [None, Some(address_with(true)), Some(address_with(false))] {
758                for nonce_even in [true, false] {
759                    let presig = S::rand(&mut rng);
760                    let public_presig = G::generator() * presig;
761                    let presig_shares = mock_shares(&mut rng, presig, t, n);
762                    let beacon = loop {
763                        let beacon = S::rand(&mut rng);
764                        if has_even_y(&(public_presig + G::generator() * beacon)) == nonce_even {
765                            break beacon;
766                        }
767                    };
768
769                    let partial_signatures = (0..n as usize)
770                        .flat_map(|i| {
771                            generate_partial_signatures(
772                                message,
773                                (vec![presig_shares[i].value], public_presig),
774                                &beacon,
775                                &avss::SharesForNode {
776                                    shares: vec![sk_shares[i].clone()],
777                                },
778                                &vk,
779                                address.as_ref(),
780                            )
781                            .unwrap()
782                            .1
783                        })
784                        .collect_vec();
785                    let signature = aggregate_signatures(
786                        message,
787                        &public_presig,
788                        &beacon,
789                        &partial_signatures,
790                        t,
791                        &vk,
792                        address.as_ref(),
793                    )
794                    .unwrap();
795
796                    match address {
797                        Some(address) => derive_verifying_key(&vk, &address).unwrap(),
798                        None => SchnorrPublicKey::try_from(&vk).unwrap(),
799                    }
800                    .verify(message, &signature)
801                    .unwrap();
802                }
803            }
804        }
805    }

815    fn test_derived_signing() {
816        let f = 2;
817        let t = f + 1;
818        let n = 3 * f + 1;
819
820        let mut rng = rand::thread_rng();
821
822        // Mock DKG
823        // Here, we don't assume anything about the parity of the vk's Y coordinate since we can't do that in a real DKG.
824        let sk_element = S::rand(&mut rng);
825        let vk_element = G::generator() * sk_element;
826
827        let sk_shares = mock_shares(&mut rng, sk_element, t, n);
828
829        // Mock nonce generation
830        let batch_size_per_weight: u16 = 100;
831        let nonces_for_dealer = (0..n)
832            .map(|_| {
833                let nonces = (0..batch_size_per_weight)
834                    .map(|_| S::rand(&mut rng))
835                    .collect_vec();
836                let public_keys = nonces.iter().map(|s| G::generator() * s).collect_vec();
837                let nonce_shares: Vec<Vec<S>> = nonces
838                    .iter()
839                    .map(|&nonce| {
840                        mock_shares(&mut rng, nonce, t, n)
841                            .iter()
842                            .map(|s| s.value)
843                            .collect_vec()
844                    })
845                    .collect_vec();
846                (nonces, public_keys, nonce_shares)
847            })
848            .collect_vec();
849
850        let outputs = (0..n)
851            .map(|i| {
852                (0..n as usize)
853                    .map(|j| {
854                        batch_avss_avid::ReceiverOutput {
855                            my_shares: SharesForNode {
856                                shares: vec![ShareBatch {

859                                        .collect_vec(),

868            .collect_vec();

895                    presigning.next().unwrap(),

901                .unwrap()
902            })
903            .collect_vec();

914            &public,
915            &beacon_value,

917                .iter()
918                .flat_map(|(_, sigs)| sigs.clone())
919                .collect_vec(),
920            t,
921            &vk_element,
922            Some(&address),
923        )
