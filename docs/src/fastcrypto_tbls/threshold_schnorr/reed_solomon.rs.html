13/// Decoder for Reed-Solomon codes.
14/// This can correct up to (d-1)/2 errors, where d is the distance of the code.
15/// The code is defined by the evaluation points `a` and the message length `k`.
16/// The distance is given by `n - k + 1`, where `n` is the length of `a`.
17///
18/// The implementation follows the Gao decoding algorithm
19/// (see <https://www.math.clemson.edu/~sgao/papers/RS.pdf>).
20pub struct RSDecoder {
21    g0: Poly<S>,
22    a: Vec<ShareIndex>,
23    k: usize,
24}
25
26impl RSDecoder {
27    /// Create a new Gao decoder with the given evaluation points `a` and message length `k`.
28    pub fn new(a: Vec<ShareIndex>, k: usize) -> Self {
29        assert!(k < a.len(), "Message length must be less than block length");
30        let mut g0 = Poly::one();
31        for ai in &a {
32            g0 *= MonicLinear(-to_scalar::<S>(ai));
33        }
34        Self { g0, a, k }
35    }
36
37    /// The length of the code words.
38    fn block_length(&self) -> usize {
39        self.a.len()
40    }
41
42    /// The length of the messages.
43    fn message_length(&self) -> usize {
44        self.k
45    }
46
47    /// The distance of the code.
48    fn distance(&self) -> usize {
49        self.block_length() - self.message_length() + 1
50    }
51
52    /// Compute the message polynomial.
53    /// Returns an error if the input length is wrong or if there are too many errors to correct.
54    pub fn compute_message_polynomial(&self, code_word: &[S]) -> FastCryptoResult<Poly<S>> {
55        // The implementation follows Algorithm 1 in Gao's paper.
56
57        if code_word.len() != self.block_length() {
58            return Err(InputLengthWrong(self.block_length()));
59        }
60
61        // Step 1: Interpolation
62        let g1 = Poly::interpolate(
63            &self
64                .a
65                .iter()
66                .zip(code_word)
67                .map(|(&index, &value)| Eval { index, value })
68                .collect_vec(),
69        )?;
70
71        // Step 2: Partial GCD
72        let (g, _, v) = Poly::partial_extended_gcd(
73            &self.g0,
74            &g1,
75            (self.message_length() + self.block_length()) / 2,
76        )?;
77
78        // Step 3: Long division
79        let (f1, r) = g.div_rem(&v)?;
80        if !r.is_zero() || f1.degree() >= self.k {
81            return Err(TooManyErrors((self.distance() - 1) / 2));
82        }
83        Ok(f1)
84    }
85
86    /// Encode the message using the Reed-Solomon code defined by the evaluation points `a`.
87    /// Returns an error if the message length is wrong.
88    pub fn encode(&self, message: Vec<S>) -> FastCryptoResult<Vec<S>> {
89        if message.len() != self.message_length() {
90            return Err(InputLengthWrong(self.message_length()));
91        }
92        let f = Poly::from(message);
93        Ok(self.a.iter().map(|&ai| f.eval(ai).value).collect_vec())
94    }
95
96    /// Try to correct the input and return the decoded message.
97    /// Returns an error if the input length is wrong or if there are too many errors to correct.
98    pub fn decode(&self, input: &[S]) -> FastCryptoResult<Vec<S>> {
99        let mut f1 = self.compute_message_polynomial(input)?.to_vec();
100        f1.truncate(self.k);
101        Ok(f1)
102    }
103}

105/// A wrapper struct for the Reed-Solomon erasure coding library.
106pub struct ErasureCoder(ReedSolomon);
107
108/// An element of `GF(2^16)` as represented by the underlying coder.
109type Element = [u8; ELEMENT_SIZE_IN_BYTES];
110
111/// Size in bytes of one `GF(2^16)` element.
112const ELEMENT_SIZE_IN_BYTES: usize = 2;
113
114/// Size in bytes of a `u32`, the width of the length prefix framing an encoded payload.
115const U32_SIZE: usize = std::mem::size_of::<u32>();
116
117#[derive(Clone, Debug, Eq, PartialEq, Serialize, Deserialize)]
118#[serde(transparent)]
119pub struct Shard(pub(crate) Vec<u8>);
120
121pub type Shards = Vec<Shard>;
122
123impl ErasureCoder {
124    /// Create a new erasure encoder/decoder.
125    ///
126    /// # Parameters
127    /// - `n`: Total number of shards.
128    /// - `k`: Number of data shards.
129    ///
130    /// # Errors
131    /// Returns [`FastCryptoError::InvalidInput`] if `k == 0`, `n <= k` or `n > 65536`.
132    pub fn new(n: usize, k: usize) -> FastCryptoResult<Self> {
133        // The code is defined over GF(2^16), which has 2^16 = 65536 elements; n cannot exceed
134        // that or the evaluation points would collide.
135        Self::check_parameters(n, k)?;
136        ReedSolomon::new(k, n - k)
137            .map_err(|_| InvalidInput)
138            .map(Self)
139    }
140
141    pub fn check_parameters(n: usize, k: usize) -> FastCryptoResult<()> {
142        if k == 0 || n <= k || n > 65536 {
143            return Err(InvalidInput);
144        }
145        Ok(())
146    }
147
148    /// Encode `data` into `n` shards of equal size, the first `k` of which hold the (zero-padded)
149    /// data and the remaining `n - k` parity. Any `k` shards suffice to reconstruct the data.
150    pub fn encode(&self, data: &[u8]) -> FastCryptoResult<Shards> {
151        if data.is_empty() {
152            return Err(InvalidInput);
153        }
154        let mut shards = self.bytes_to_element_shards(data)?;
155        self.0.encode(&mut shards).map_err(|_| InvalidInput)?;
156        Ok(shards.into_iter().map(|s| Shard(s.concat())).collect_vec())
157    }
158
159    /// Reconstruct the original data from `n` (possibly missing) shards. Fails if more than
160    /// `n - k` shards are missing, if the present shards are inconsistent with any single
161    /// codeword, or if the recovered length prefix doesn't fit the recovered bytes.
162    pub fn decode(&self, shards: Vec<Option<Shard>>) -> FastCryptoResult<Vec<u8>> {
163        if shards.len() != self.0.total_shard_count() {
164            return Err(InputLengthWrong(self.0.total_shard_count()));
165        }
166
167        let mut shards: Vec<Option<Vec<Element>>> = shards
168            .into_iter()
169            .map(|opt| {
170                opt.map(|Shard(bytes)| bytes_to_elements(&bytes))
171                    .transpose()
172            })
173            .collect::<FastCryptoResult<_>>()?;
174        self.0.reconstruct(&mut shards).map_err(|_| InvalidInput)?;
175        let shards = shards
176            .into_iter()
177            .map(|s| s.ok_or(InvalidInput))
178            .collect::<FastCryptoResult<Vec<_>>>()?;
179
180        // Ensure the reconstructed shards are consistent
181        if !self.0.verify(&shards).map_err(|_| InvalidInput)? {
182            return Err(TooManyErrors(0)); // This is just an erasure code, so we can't correct errors.
183        }
184
185        self.element_shards_to_bytes(shards)
186    }
187
188    /// An injective mapping from arbitrary binary `data` to `Vec<Vec<Element>>`, inverted by
189    /// [`Self::element_shards_to_bytes`]. Returns [`InvalidInput`] if `data.len()` does not fit in a
190    /// `u32`.
191    fn bytes_to_element_shards(&self, data: &[u8]) -> FastCryptoResult<Vec<Vec<Element>>> {
192        // Data format: a little-endian u32 length prefix followed by `data`, then zero padding to
193        // fill a whole number of `Element`s across all shards.
194        let len = u32::try_from(data.len()).map_err(|_| InvalidInput)?;
195        let framed_len = U32_SIZE + data.len();
196        let shard_size = framed_len.div_ceil(ELEMENT_SIZE_IN_BYTES * self.0.data_shard_count());
197        let total = ELEMENT_SIZE_IN_BYTES * shard_size * self.0.total_shard_count();
198        Ok(len
199            .to_le_bytes()
200            .into_iter()
201            .chain(data.iter().copied())
202            .chain(std::iter::repeat(0))
203            .take(total)
204            // Each `Element` is two bytes.
205            .tuples::<(u8, u8)>()
206            .map(|(a, b)| [a, b])
207            .chunks(shard_size)
208            .into_iter()
209            .map(|shard| shard.collect_vec())
210            .collect())
211    }
212
213    /// Recover the data from the reconstructed `shards`. Returns [`InvalidInput`] if the encoding
214    /// is invalid.
215    fn element_shards_to_bytes(&self, shards: Vec<Vec<Element>>) -> FastCryptoResult<Vec<u8>> {
216        // A valid codeword has exactly `total_shard_count` shards, all of the same element length.
217        if shards.len() != self.0.total_shard_count()
218            || get_uniform_value(shards.iter().map(Vec::len)).is_none()
219        {
220            return Err(InvalidInput);
221        }
222        let framed: Vec<u8> = shards
223            .into_iter()
224            .take(self.0.data_shard_count())
225            .flatten()
226            .flatten()
227            .collect();
228        let prefix = framed.get(..U32_SIZE).ok_or(InvalidInput)?;
229        let len = u32::from_le_bytes(prefix.try_into().unwrap()) as usize;
230        let end = U32_SIZE.checked_add(len).ok_or(InvalidInput)?;
231        if framed
232            .get(end..)
233            .ok_or(InvalidInput)?
234            .iter()
235            .any(|&b| b != 0)
236        {
237            return Err(InvalidInput);
238        }
239        Ok(framed[U32_SIZE..end].to_vec())
240    }
241}

243/// Reinterpret `bytes` as a sequence of [Element]s. Fails with [`InvalidInput`] if the input
244/// length is not a multiple of [`ELEMENT_SIZE_IN_BYTES`].
245fn bytes_to_elements(bytes: &[u8]) -> FastCryptoResult<Vec<Element>> {
246    if !bytes.len().is_multiple_of(ELEMENT_SIZE_IN_BYTES) {
247        return Err(InvalidInput);
248    }
249    Ok(bytes
250        .chunks_exact(ELEMENT_SIZE_IN_BYTES)
251        .map(|p| p.try_into().expect("chunk has ELEMENT_SIZE_IN_BYTES bytes"))
252        .collect())
253}

259    #[test]
260    fn test_gao_decoder() {
261        let a = (1..=7).map(|i| ShareIndex::new(i).unwrap()).collect_vec();
262        let k = 3;
263        let decoder = RSDecoder::new(a.clone(), k);
264
265        let message = vec![S::from(11u128), S::from(22u128), S::from(33u128)];
266        let code_word = decoder.encode(message.clone()).unwrap();
267
268        // Introduce errors
269        let mut received = code_word.clone();
270        received[4] = S::from(20u128); // Error at position 4
271        received[2] = S::from(200u128); // Error at position 2
272
273        let decoded_message = decoder.decode(&received).unwrap();
274        assert_eq!(decoded_message, message);
275
276        // Test with too many errors
277        let mut received = code_word.clone();
278        received[4] = S::from(20u128); // Error at position 4
279        received[3] = S::from(2000u128); // Error at position 3
280        received[2] = S::from(200u128); // Error at position 2
281        assert!(decoder.decode(&received).is_err());
282    }

292    #[test]
293    fn test_erasure_coder_roundtrip() {
294        let n = 10;
295        let k = 6;
296        let coder = ErasureCoder::new(n, k).unwrap();
297
298        for len in [1usize, 2, 3, 7, 8, 31, 32, 33, 100, 255] {
299            let data: Vec<u8> = (0..len)
300                .map(|i| (i as u8).wrapping_mul(31).wrapping_add(7))
301                .collect();
302            let shards = coder.encode(&data).unwrap();
303            assert_eq!(shards.len(), n);
304
305            // Remove up to `parity` shards (erasures) and reconstruct.
306            let mut opt_shards: Vec<Option<Shard>> = shards.into_iter().map(Some).collect();
307            for shard in opt_shards.iter_mut().take(n - k) {
308                *shard = None;
309            }
310
311            let coder = ErasureCoder::new(n, k).unwrap();
312            let recovered = coder.decode(opt_shards).unwrap();
313            assert_eq!(recovered, data);
314        }
315    }

350    #[test]
351    fn test_erasure_coder_encode_shard_lengths() {
352        // Each GF(2^16) element is 2 bytes; the framed payload is `u32_len || data` so the
353        // pre-padding length is `4 + data_len`. Shards are sized to a whole number of pairs with
354        // pair count ⌈(4 + data_len) / (2 · k)⌉.
355        for &(n, k, data_len, expected_shard_bytes) in &[
356            (10, 6, 1, 2),       // ⌈  5 /  12⌉ = 1 pair
357            (10, 6, 8, 2),       // ⌈ 12 /  12⌉ = 1 pair
358            (10, 6, 9, 4),       // ⌈ 13 /  12⌉ = 2 pairs
359            (10, 6, 12, 4),      // ⌈ 16 /  12⌉ = 2 pairs
360            (10, 6, 100, 18),    // ⌈104 /  12⌉ = 9 pairs
361            (800, 268, 2028, 8), // ⌈2032 / 536⌉ = 4 pairs
362        ] {
363            let coder = ErasureCoder::new(n, k).unwrap();
364            let data: Vec<u8> = (0..data_len).map(|i| i as u8).collect();
365            let shards = coder.encode(&data).unwrap();
366            assert_eq!(shards.len(), n, "shard count");
367            for shard in &shards {
368                assert_eq!(
369                    shard.0.len(),
370                    expected_shard_bytes,
371                    "shard byte length for n={n}, k={k}, data_len={data_len}"
372                );
373            }
374        }
375    }

377    #[test]
378    fn test_element_shards_to_bytes_validation() {
379        let coder = ErasureCoder::new(10, 6).unwrap();
380
381        // One `Element` per shard so all shards have the same length. Flattening the first
382        // `k = 6` shards yields `u32_le(1) || data_byte || pad || zero padding`.
383        let data_byte = 0xABu8;
384        let make_shards = |pad: u8| -> Vec<Vec<Element>> {
385            let mut shards: Vec<Vec<Element>> =
386                vec![vec![[1, 0]], vec![[0, 0]], vec![[data_byte, pad]]];
387            shards.resize(coder.0.total_shard_count(), vec![[0, 0]]);
388            shards
389        };
390
391        // Well-formed with zero padding decodes; non-zero padding is rejected.
392        assert_eq!(
393            coder.element_shards_to_bytes(make_shards(0)).unwrap(),
394            vec![data_byte]
395        );
396        assert!(matches!(
397            coder.element_shards_to_bytes(make_shards(1)),
398            Err(InvalidInput)
399        ));
400
401        // Wrong shard count is rejected.
402        let mut too_few = make_shards(0);
403        too_few.pop();
404        assert!(matches!(
405            coder.element_shards_to_bytes(too_few),
406            Err(InvalidInput)
407        ));
408
409        // Shards of differing lengths are rejected.
410        let mut ragged = make_shards(0);
411        ragged[0].push([0, 0]);
412        assert!(matches!(
413            coder.element_shards_to_bytes(ragged),
414            Err(InvalidInput)
415        ));
416    }
