12/// An iterator that yields presigning tuples (t_i, p_i).
13pub struct Presignatures {
14    secret: Vec<LazyPascalMatrixMultiplier<S>>,
15    public: LazyPascalMatrixMultiplier<G>,
16}
17
18impl Iterator for Presignatures {
19    type Item = (Vec<S>, G);
20
21    fn next(&mut self) -> Option<Self::Item> {
22        // `public` drives the length; `secret` is empty for a zero-weight party.
23        let public = self.public.next()?;
24        let secret = self
25            .secret
26            .iter_mut()
27            .map(Iterator::next)
28            .collect::<Option<Vec<_>>>()
29            .expect("secret and public multipliers have equal length");
30        Some((secret, public))
31    }
32
33    fn size_hint(&self) -> (usize, Option<usize>) {
34        let remaining = self.public.len();
35        (remaining, Some(remaining))
36    }
37}
38
39impl ExactSizeIterator for Presignatures {}

41impl Presignatures {
42    /// Based on the output of a batched AVSS from multiple dealers, create a presignature
43    /// generator.
44    ///
45    /// All parties must use the same outputs in the same order, and the output from a dealer with
46    /// weight `w` should be equal to `batch_size_per_weight * w`. The outputs must come from
47    /// distinct dealers, with at most one output per dealer.
48    ///
49    /// More parties contributing outputs gives more presignatures, so include as many as possible
50    /// but at least `params.t` (by weight).
51    /// Caller should wait for at least `params.t` outputs, plus some \delta time to collect more.
52    ///
53    /// `params.t` is the reconstruction threshold. The nonce polynomials are shared at degree
54    /// `params.t - 1`, so this produces `total_weight - (params.t - 1)` presignatures per nonce
55    /// position: the privacy threshold of the sharings is `t - 1`, meaning a sub-`t` coalition can
56    /// know or bias up to `t - 1` of the input nonces, so only `total_weight - (t - 1)` combined
57    /// nonces per position remain uniformly random and safe to output.
58    ///
59    /// An InvalidInput error will be returned if:
60    /// * `params.t` is zero,
61    /// * The total weight of the dealers for the outputs is not at least `params.t`,
62    /// * The batch size of one of the outputs is not divisible by `batch_size_per_weight`,
63    /// * The same output is given more than once,
64    /// * or if batch_size_per_weight is zero.
65    pub fn new(
66        outputs: Vec<ReceiverOutput>,
67        batch_size_per_weight: u16,
68        params: Parameters,
69        use_legacy: bool,
70    ) -> FastCryptoResult<Self> {
71        if batch_size_per_weight == 0 {
72            return Err(InvalidInput);
73        }
74        let batch_size_per_weight = batch_size_per_weight as usize;
75
76        // Reject duplicate outputs. Outputs from different dealings have different public keys with
77        // overwhelming probability. Outputs without public keys add no weight and are ignored.
78        if outputs
79            .iter()
80            .filter(|o| !o.public_keys.is_empty())
81            .tuple_combinations()
82            .any(|(a, b)| a.public_keys == b.public_keys)
83        {
84            return Err(InvalidInput);
85        }
86
87        // Recover each dealer's weight from its public key count, which works even for a
88        // zero-weight party.
89        let weights = outputs
90            .iter()
91            .map(|o| {
92                let batch_size = o.public_keys.len();
93                (batch_size % batch_size_per_weight == 0)
94                    .then_some(batch_size / batch_size_per_weight)
95                    .ok_or(InvalidInput)
96            })
97            .collect::<FastCryptoResult<Vec<_>>>()?;
98        let total_weight_of_outputs: usize = weights.iter().sum();
99        if params.t == 0 || total_weight_of_outputs < params.t as usize {
100            return Err(InvalidInput);
101        }
102
103        // TODO: remove legacy mode once the old protocol is deprecated.
104        let height = if use_legacy {
105            total_weight_of_outputs - params.f as usize
106        } else {
107            total_weight_of_outputs - (params.t as usize - 1)
108        };
109
110        // This party's weight, aka its number of shares
111        let my_weight =
112            get_uniform_value(outputs.iter().map(|o| o.my_shares.weight())).ok_or(InvalidInput)?;
113
114        // Each share's batch must cover exactly the nonces dealt by that dealer.
115        // The zero-weight party holds no shares, so there is nothing to check.
116        if outputs.iter().zip(weights.iter()).any(|(o, w)| {
117            !o.my_shares.shares.is_empty()
118                && o.my_shares
119                    .try_uniform_batch_size()
120                    .ok()
121                    .is_none_or(|bs| bs != *w * batch_size_per_weight)
122        }) {
123            return Err(InvalidInput);
124        }
125
126        // There is one secret presigning output per shares for this party
127        let secret = (0..my_weight as usize)
128            .map(|i| {
129                LazyPascalMatrixMultiplier::new(
130                    height,
131                    (0..batch_size_per_weight)
132                        .map(|j| {
133                            outputs
134                                .iter()
135                                .zip(weights.iter())
136                                .flat_map(|(o, w)| {
137                                    o.my_shares.shares[i].batch[j * w..(j + 1) * w].to_vec()
138                                })
139                                .collect()
140                        })
141                        .collect(),
142                )
143            })
144            .collect_vec();
145
146        let public = LazyPascalMatrixMultiplier::new(
147            height,
148            (0..batch_size_per_weight)
149                .map(|j| {
150                    outputs
151                        .iter()
152                        .zip(weights.iter())
153                        .flat_map(|(o, w)| o.public_keys[j * w..(j + 1) * w].to_vec())
154                        .collect()
155                })
156                .collect_vec(),
157        );
158
159        // Sanity check that the multiplier sizes match the expected nonce count.
160        let expected_len = height * batch_size_per_weight;
161        assert!(secret.iter().all(|s| s.len() == expected_len));
162        assert_eq!(public.len(), expected_len);
163
164        Ok(Self { secret, public })
165    }
166}

169mod tests {
170    use super::Presignatures;
171    use crate::threshold_schnorr::batch_avss::{ReceiverOutput, ShareBatch, SharesForNode};
172    use crate::threshold_schnorr::{Parameters, G, S};

176    #[test]
177    fn test_new_with_zero_weight_party() {
178        // A zero-weight party gets ReceiverOutputs with empty shares; this must not panic.
179        let batch_size_per_weight: u16 = 2;
180        let params = Parameters { t: 2, f: 1 }; // total weight is 2; requires t > f
181
182        // Two weight-1 dealers: each output has batch_size_per_weight public keys, no shares.
183        let outputs = (0..2)
184            .map(|i| ReceiverOutput {
185                my_shares: SharesForNode { shares: vec![] },
186                public_keys: vec![G::generator() * S::from(i + 1); batch_size_per_weight as usize],
187            })
188            .collect::<Vec<_>>();
189
190        let presignatures =
191            Presignatures::new(outputs, batch_size_per_weight, params, false).unwrap();
192
193        let total_weight_of_outputs = 2;
194        let expected_len =
195            (total_weight_of_outputs - (params.t as usize - 1)) * batch_size_per_weight as usize;
196        assert_eq!(presignatures.len(), expected_len);
197
198        let tuples = presignatures.collect::<Vec<_>>();
199        assert_eq!(tuples.len(), expected_len);
200        assert!(tuples.iter().all(|(secret, _public)| secret.is_empty()));
201    }

203    #[test]
204    fn test_presig_count_uses_privacy_threshold_not_f() {
205        // Regression test for the SI-matrix height: in the default (non-legacy) mode it must be
206        // `total_weight - (t - 1)`, the privacy threshold of the degree-`(t-1)` nonce sharings,
207        // NOT `total_weight - f`. The two differ exactly when `t > f + 1`, so pick such a case:
208        // t = 3, f = 1 (t - 1 = 2 != f = 1). The legacy mode still uses `total_weight - f`.
209        let batch_size_per_weight: u16 = 2;
210        let params = Parameters { t: 3, f: 1 };
211
212        // Four weight-1 dealers -> total weight 4. Zero-weight receiver perspective (empty shares).
213        let outputs = (0..4)
214            .map(|i| ReceiverOutput {
215                my_shares: SharesForNode { shares: vec![] },
216                public_keys: vec![G::generator() * S::from(i + 1); batch_size_per_weight as usize],
217            })
218            .collect::<Vec<_>>();
219
220        // Default mode (privacy threshold t-1): (4 - (3 - 1)) * 2 = 4.
221        let new =
222            Presignatures::new(outputs.clone(), batch_size_per_weight, params, false).unwrap();
223        assert_eq!(
224            new.len(),
225            (4 - (params.t as usize - 1)) * batch_size_per_weight as usize
226        );
227        assert_eq!(new.len(), 4);
228
229        // Legacy mode (total_weight - f): (4 - 1) * 2 = 6.
230        let legacy = Presignatures::new(outputs, batch_size_per_weight, params, true).unwrap();
231        assert_eq!(
232            legacy.len(),
233            (4 - params.f as usize) * batch_size_per_weight as usize
234        );
235        assert_eq!(legacy.len(), 6);
236    }

238    #[test]
239    fn test_new_rejects_too_short_batch() {
240        // Each dealer deals batch_size_per_weight nonces per weight, so a weight-1 dealer's share
241        // batch must have batch_size_per_weight entries. A shorter batch must be rejected, not
242        // panic.
243        let batch_size_per_weight: u16 = 2;
244        let params = Parameters { t: 2, f: 1 }; // total weight is 2; requires t > f
245
246        let outputs = (0..2)
247            .map(|i| ReceiverOutput {
248                my_shares: SharesForNode {
249                    shares: vec![ShareBatch {
250                        index: ShareIndex::new(1).unwrap(),
251                        batch: vec![S::generator()], // length 1 < expected 2
252                        blinding_share: S::generator(),
253                    }],
254                },
255                public_keys: vec![G::generator() * S::from(i + 1); batch_size_per_weight as usize],
256            })
257            .collect::<Vec<_>>();
258
259        assert!(Presignatures::new(outputs, batch_size_per_weight, params, false).is_err());
260    }
